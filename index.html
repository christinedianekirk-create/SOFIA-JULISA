// Cambiar de ventana al hacer clic en el menú lateral const menuItems = document.querySelectorAll(".menu-item"); const windows = document.querySelectorAll(".window"); menuItems.forEach((item) => { item.addEventListener("click", () => { const targetId = item.getAttribute("data-target"); // Quitar "active" de todos los botones menuItems.forEach((btn) => btn.classList.remove("active")); // Poner "active" al botón actual item.classList.add("active"); // Ocultar todas las ventanas windows.forEach((win) => win.classList.remove("active")); // Mostrar solo la ventana seleccionada const targetWindow = document.getElementById(targetId); if (targetWindow) { targetWindow.classList.add("active"); } }); });
